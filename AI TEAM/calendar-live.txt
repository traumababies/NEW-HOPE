        if (r.theme && !themeEl.value.trim()) { filled.theme = themeEl.value; themeEl.value = r.theme; }
        const noteBits = [r.theme_ideas, r.creative_direction].filter(Boolean).join('\n\n');
        if (noteBits && !notesEl.value.trim()) { filled.thoughts = notesEl.value; notesEl.value = noteBits; }
        if (!Object.keys(filled).length) { msg.textContent = 'Nothing to add — already filled in.'; filled = null; return; }
        msg.textContent = 'Sol filled it. Save to keep it, or Delete Sol’s.';
      } catch (e) {
        msg.textContent = 'Request failed: ' + e;
      } finally {
        runBtn.disabled = false;
      }
    });

    clearBtn.addEventListener('click', () => {
      themeEl.value = '';
      notesEl.value = '';
      filled = null;
      msg.textContent = 'Cleared. Write your own, then Save.';
    });
  });
})();
